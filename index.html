function setRoute(type) { if (control) { map.removeControl(control); } if (type === "adr") { // Adrénaline : parcours sinueux dans le Vexin (L'Isle-Adam -> Vallangoujard -> Marines) control = L.Routing.control({ waypoints: [ L.latLng(49.1083, 2.2244), // L’Isle-Adam L.latLng(49.1168, 2.1144), // Vallangoujard (route D151) L.latLng(49.1381, 1.9814) // Marines ], lineOptions: { styles: [{ color: "#e74c3c", weight: 5 }] }, routeWhileDragging: false, draggableWaypoints: false, addWaypoints: false, show: false }).addTo(map); } if (type === "lib") { // Liberté : Cergy → Enghien-les-Bains control = L.Routing.control({ waypoints: [ L.latLng(49.0356, 2.0600), L.latLng(48.9726, 2.3166) ], lineOptions: { styles: [{ color: "#3498db", weight: 5 }] }, routeWhileDragging: false, draggableWaypoints: false, addWaypoints: false, show: false }).addTo(map); } if (type === "det") { // Détente : Magny-en-Vexin → Auvers-sur-Oise control = L.Routing.control({ waypoints: [ L.latLng(49.1553, 1.7844), L.latLng(49.1252, 2.0555), L.latLng(49.0712, 2.1630) ], lineOptions: { styles: [{ color: "#27ae60", weight: 5 }] }, routeWhileDragging: false, draggableWaypoints: false, addWaypoints: false, show: false }).addTo(map); } }
